Verify tables:

Verify after install:
  kubectl exec -n {{ .Release.Namespace }} statefulset/{{ include "clickhouse.fullname" . }} -- \
    clickhouse-client --query "SELECT name, value FROM system.settings WHERE name LIKE '%lightweight_update%'"
